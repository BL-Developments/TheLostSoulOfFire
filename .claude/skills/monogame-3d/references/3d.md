## BasicEffect initialization

```csharp
// In LoadContent:
_basicEffect = new BasicEffect(GraphicsDevice);

// Option A — vertex colors only
_basicEffect.VertexColorEnabled  = true;
_basicEffect.LightingEnabled     = false;
_basicEffect.TextureEnabled      = false;

// Option B — texture only
_basicEffect.TextureEnabled      = true;
_basicEffect.VertexColorEnabled  = false;
_basicEffect.Texture             = _myTexture;

// Option C — texture + vertex color tint
_basicEffect.TextureEnabled      = true;
_basicEffect.VertexColorEnabled  = true;   // Color tints the texture sample

// Option D — lighting
_basicEffect.LightingEnabled          = true;
_basicEffect.AmbientLightColor        = new Vector3(0.2f, 0.2f, 0.2f);
_basicEffect.DirectionalLight0.Enabled      = true;
_basicEffect.DirectionalLight0.Direction    = Vector3.Normalize(new Vector3(1, -1, -1));
_basicEffect.DirectionalLight0.DiffuseColor = Vector3.One;
// Or call shorthand:
_basicEffect.EnableDefaultLighting();
```

## World / View / Projection matrices

```csharp
// World — transforms object from model space to world space
_basicEffect.World = Matrix.CreateRotationY(_angle) *
                     Matrix.CreateTranslation(_position);

// View — camera position and orientation
_basicEffect.View = Matrix.CreateLookAt(
    new Vector3(0, 0, 10),   // camera position
    Vector3.Zero,            // target (look-at point)
    Vector3.Up               // up vector
);

// Projection — perspective frustum
_basicEffect.Projection = Matrix.CreatePerspectiveFieldOfView(
    MathHelper.ToRadians(45f),
    GraphicsDevice.Viewport.AspectRatio,
    0.1f,    // near clip
    1000f    // far clip
);
```

Set `World` per object before each `pass.Apply()`. `View` and `Projection` typically stay constant for a frame.

## 3D camera — rotate and move

```csharp
// Fields:
private Vector3 _cameraPos     = new Vector3(0, 0, 10f);
private float   _cameraYaw     = 0f;
private const float MoveSpeed  = 5f;
private const float TurnSpeed  = 1.5f;

// In Update:
float elapsed = (float)gameTime.ElapsedGameTime.TotalSeconds;

// Rotate (y-axis / yaw):
if (Keyboard.GetState().IsKeyDown(Keys.Left))  _cameraYaw += TurnSpeed * elapsed;
if (Keyboard.GetState().IsKeyDown(Keys.Right)) _cameraYaw -= TurnSpeed * elapsed;

// Compute forward direction from yaw:
Vector3 forward = Vector3.Transform(Vector3.Forward,
    Matrix.CreateRotationY(_cameraYaw));

// Move forward/backward:
if (Keyboard.GetState().IsKeyDown(Keys.Up))   _cameraPos += forward * MoveSpeed * elapsed;
if (Keyboard.GetState().IsKeyDown(Keys.Down))  _cameraPos -= forward * MoveSpeed * elapsed;

// Rebuild view matrix:
_view = Matrix.CreateLookAt(_cameraPos, _cameraPos + forward, Vector3.Up);
```

## Fit camera to scene

```csharp
// Merge bounding spheres of all objects in the scene:
BoundingSphere scene = BoundingSphere.CreateFromBoundingBox(
    BoundingBox.CreateFromSphere(objects[0].BoundingSphere));

foreach (var obj in objects)
    scene = BoundingSphere.CreateMerged(scene, obj.BoundingSphere);

// Distance required to contain the full sphere within the FOV:
float fovHalfY = fovY / 2f;
float dist     = scene.Radius / MathF.Sin(fovHalfY);

// Move camera along its current backward direction:
Matrix viewInverse  = Matrix.Invert(_view);
Vector3 backward    = viewInverse.Backward;
_cameraPos          = scene.Center + backward * dist;

_view = Matrix.CreateLookAt(_cameraPos, scene.Center, Vector3.Up);
```

## Custom vertex type (IVertexType)

```csharp
public struct MyVertex : IVertexType
{
    public Vector3 Position;        // offset 0  (12 bytes)
    public Vector2 TexCoord;        // offset 12 ( 8 bytes)
    // Add more fields here, update offsets and VertexDeclaration accordingly

    public static readonly VertexDeclaration VertexDeclaration = new VertexDeclaration(
        new VertexElement( 0, VertexElementFormat.Vector3, VertexElementUsage.Position,          0),
        new VertexElement(12, VertexElementFormat.Vector2, VertexElementUsage.TextureCoordinate, 0)
    );

    VertexDeclaration IVertexType.VertexDeclaration => VertexDeclaration;
}

// Spinning cube example — build vertex buffer once in LoadContent:
_vertexBuffer = new VertexBuffer(GraphicsDevice, MyVertex.VertexDeclaration,
    vertices.Length, BufferUsage.WriteOnly);
_vertexBuffer.SetData(vertices);
```

Common `VertexElementFormat` sizes: `Vector2` = 8 B, `Vector3` = 12 B, `Vector4` = 16 B, `Color` = 4 B.

## DrawUserPrimitives patterns

Immediate rendering — no buffer required, vertices reside in managed memory:

```csharp
var triangle = new VertexPositionColor[]
{
    new VertexPositionColor(new Vector3( 0,  1, 0), Color.Red),
    new VertexPositionColor(new Vector3(-1, -1, 0), Color.Green),
    new VertexPositionColor(new Vector3( 1, -1, 0), Color.Blue),
};

foreach (EffectPass pass in _basicEffect.CurrentTechnique.Passes)
{
    pass.Apply();
    GraphicsDevice.DrawUserPrimitives(
        PrimitiveType.TriangleList,
        triangle,
        0,            // vertex offset
        1             // primitive count (triangles)
    );
}
```

Primitive count formula:
- TriangleList: `vertexCount / 3`
- TriangleStrip: `vertexCount - 2`
- LineList: `vertexCount / 2`
- LineStrip: `vertexCount - 1`

## DrawUserIndexedPrimitives patterns

Use indices to share vertices across primitives:

```csharp
// A quad (2 triangles) with 4 vertices:
var verts = new VertexPositionColor[]
{
    new VertexPositionColor(new Vector3(-1,  1, 0), Color.Red),    // 0: top-left
    new VertexPositionColor(new Vector3( 1,  1, 0), Color.Green),  // 1: top-right
    new VertexPositionColor(new Vector3(-1, -1, 0), Color.Blue),   // 2: bottom-left
    new VertexPositionColor(new Vector3( 1, -1, 0), Color.White),  // 3: bottom-right
};

var indices = new short[] { 0, 1, 2, 1, 3, 2 }; // 2 triangles

foreach (EffectPass pass in _basicEffect.CurrentTechnique.Passes)
{
    pass.Apply();
    GraphicsDevice.DrawUserIndexedPrimitives(
        PrimitiveType.TriangleList,
        verts,    0, 4,    // vertices, vertexOffset, numVertices
        indices,  0, 2     // indices,  indexOffset,  primitiveCount
    );
}
```

## VertexBuffer and IndexBuffer

```csharp
// In LoadContent — static geometry:
_vb = new VertexBuffer(GraphicsDevice,
    VertexPositionColor.VertexDeclaration,
    vertices.Length, BufferUsage.WriteOnly);
_vb.SetData(vertices);

_ib = new IndexBuffer(GraphicsDevice,
    IndexElementSize.SixteenBits,    // or ThirtyTwoBits for >65535 indices
    indices.Length, BufferUsage.WriteOnly);
_ib.SetData(indices);

// In Draw:
GraphicsDevice.SetVertexBuffer(_vb);
GraphicsDevice.Indices = _ib;

foreach (EffectPass pass in _basicEffect.CurrentTechnique.Passes)
{
    pass.Apply();
    GraphicsDevice.DrawIndexedPrimitives(
        PrimitiveType.TriangleList,
        0,              // baseVertex
        0,              // startIndex
        primitiveCount
    );
}
```

## DynamicVertexBuffer

For geometry rebuilt every frame (particles, procedural meshes):

```csharp
// In LoadContent:
_dvb = new DynamicVertexBuffer(GraphicsDevice,
    VertexPositionColor.VertexDeclaration,
    maxVertices, BufferUsage.WriteOnly);

// In Update (after computing new vertex data):
_dvb.SetData(_vertices, 0, _activeVertexCount, SetDataOptions.Discard);
// SetDataOptions.Discard avoids GPU pipeline stalls

// In Draw:
GraphicsDevice.SetVertexBuffer(_dvb);
foreach (EffectPass pass in _basicEffect.CurrentTechnique.Passes)
{
    pass.Apply();
    GraphicsDevice.DrawPrimitives(PrimitiveType.TriangleList, 0, _activeVertexCount / 3);
}
```

## RasterizerState options

```csharp
// Predefined (use these when possible — no allocation):
GraphicsDevice.RasterizerState = RasterizerState.CullCounterClockwise; // default 3D
GraphicsDevice.RasterizerState = RasterizerState.CullClockwise;
GraphicsDevice.RasterizerState = RasterizerState.CullNone;

// Custom — create once in LoadContent:
var wireframe = new RasterizerState
{
    CullMode = CullMode.None,
    FillMode = FillMode.WireFrame,
    MultiSampleAntiAlias = true   // smooths wireframe edges
};
```

After drawing with a custom state, restore the default:
```csharp
GraphicsDevice.RasterizerState = RasterizerState.CullCounterClockwise;
```

## Model loading and rendering

```csharp
// In LoadContent:
_model = Content.Load<Model>("Models/MyModel");
_boneTransforms = new Matrix[_model.Bones.Count];

// In Draw:
_model.CopyAbsoluteBoneTransformsTo(_boneTransforms);

foreach (ModelMesh mesh in _model.Meshes)
{
    foreach (BasicEffect effect in mesh.Effects)
    {
        effect.World      = _boneTransforms[mesh.ParentBone.Index] * _worldMatrix;
        effect.View       = _view;
        effect.Projection = _projection;
        effect.EnableDefaultLighting();
    }
    mesh.Draw();
}
```

Move model with rotation (keyboard / gamepad):
```csharp
// In Update:
if (Keyboard.GetState().IsKeyDown(Keys.Left))  _modelRotation -= rotSpeed * elapsed;
if (Keyboard.GetState().IsKeyDown(Keys.Right)) _modelRotation += rotSpeed * elapsed;
if (Keyboard.GetState().IsKeyDown(Keys.Up))    _modelPosition += forward * moveSpeed * elapsed;

_worldMatrix = Matrix.CreateRotationY(_modelRotation) *
               Matrix.CreateTranslation(_modelPosition);
```

## Mouse picking — Viewport.Unproject

```csharp
// In Update:
MouseState ms = Mouse.GetState();
Viewport   vp = GraphicsDevice.Viewport;

// Unproject mouse from screen to world space (near and far plane):
Vector3 nearWorld = vp.Unproject(new Vector3(ms.X, ms.Y, 0f),
    _projection, _view, Matrix.Identity);
Vector3 farWorld  = vp.Unproject(new Vector3(ms.X, ms.Y, 1f),
    _projection, _view, Matrix.Identity);

var pickRay = new Ray(nearWorld, Vector3.Normalize(farWorld - nearWorld));

// Test against bounding sphere:
float? dist = pickRay.Intersects(_targetSphere);
if (dist.HasValue)
    Vector3 hitPoint = nearWorld + pickRay.Direction * dist.Value;

// Test against bounding box:
float? distBox = pickRay.Intersects(_targetBox);
```

For multiple objects, collect all hits with their distances and sort ascending — index 0 is the frontmost:
```csharp
var hits = objects
    .Select(o => (obj: o, dist: pickRay.Intersects(o.Sphere)))
    .Where(h => h.dist.HasValue)
    .OrderBy(h => h.dist.Value)
    .ToList();
```

## 3D collision — BoundingSphere per mesh

```csharp
// Mesh-level collision check between two models:
bool ModelsCollide(Model a, Matrix aWorld, Model b, Matrix bWorld)
{
    foreach (ModelMesh meshA in a.Meshes)
    {
        BoundingSphere sA = meshA.BoundingSphere.Transform(aWorld);
        foreach (ModelMesh meshB in b.Meshes)
        {
            BoundingSphere sB = meshB.BoundingSphere.Transform(bWorld);
            if (sA.Intersects(sB)) return true;
        }
    }
    return false;
}

// Collision response:
if (ModelsCollide(_modelA, _worldA, _modelB, _worldB))
{
    _velocityA = -_velocityA;
    _positionA = _prevPositionA;   // roll back to last safe position
}
```

## Transform a point with a matrix

```csharp
// Rotate a point around Y axis:
Vector3 point       = new Vector3(1, 0, 0);
Matrix  rotation    = Matrix.CreateRotationY(MathHelper.PiOver2);
Vector3 rotated     = Vector3.Transform(point, rotation);

// Combined transform (scale → rotate → translate):
Matrix world = Matrix.CreateScale(2f) *
               Matrix.CreateRotationY(angle) *
               Matrix.CreateTranslation(position);
Vector3 worldPoint = Vector3.Transform(localPoint, world);

// Transform a direction (ignores translation):
Vector3 worldDir = Vector3.TransformNormal(localDir, world);
```

Use `Vector3.TransformNormal` for direction vectors (normals, velocities) — it skips the translation component.
